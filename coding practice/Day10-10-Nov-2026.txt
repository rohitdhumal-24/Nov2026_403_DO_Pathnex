# Day 10 — Advanced Kubernetes, Jenkins, and GitLab CI

## 🔹 Ansible — Create Docker Containers for Multiple Applications

- name: Setup Docker Container for multiple apps
  hosts: all
  become: yes
  tasks: 
    - name: Pull web app image
      docker_image:
        name: my-web-app
        source: pull

    - name: Pull database app image
      docker_image: 
      name: my-database-app
      source: pull

    - name: Run web app containers
      docker_container
      image: web-app
      state: started
      published_port: "8080.80"  

    - name: Run Database App containers
      docker_container: databse-app
      state: started
      publicshed_ports: "3306:3306"        

🔹 Terraform — VPC with Private and Public Subnets
resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "public" {
  vpc_id                  = aws_subnet.main.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "us-east-1a"
  map_public_ip_on_launch = true  
}

resource "aws_subnet" "private" {
  vpc_id            = aws_vpc.main.id
  cidr_block        = "10.0.1.0/24"
  availability_zone = "us-east-1b"       
}

🔹 Jenkinsfile — Build and Deploy to Kubernetes
pipeline {
    agent any
    environment {
      IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('Push Dockr Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {


                    }
                } 
            }
        }
        stage('Deply to Kuberneties') {

🔹 GitLab CI/CD — Build, Push, and Deploy
stages:
  - build
  - push
  - deploy

build: 
  stage: build
  script:
    - docker build -t pathnex-web-app .

push: 
  stage: push
  script: 
  - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
  - docker push pathnex-web-app
