Day 09 — Docker Integration with Jenkins & GitLab
🔹 Ansible — Setup Docker Container for Nginx
# Download the Nginx image from Docker Hub.
# Create and start an Nginx container using that image.
- name: Setup docker container for nginx
  hosts: all
  become: yes
  tasks:
    - name: pull nginx image
      docker_image:
        name: nginx
        source: pull
    - name: Run nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports: "8080:80"
